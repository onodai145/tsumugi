- CSP の下で WebDriver のスクリプト実行かデバッグブリッジが動かない場合は、止めて相談する。
- 説明・コメント・文書は日本語(`CLAUDE.md` は周囲に合わせて英語)。

```rust
//! `tauri.conf.json` の CSP が緩んでいないことを確かめる(Issue #612、
//! 設計は `docs/superpowers/specs/2026-11-28-issue-612-csp-design.md`)。

use serde_json::Value;

fn security() -> Value {
    let conf: Value =
        serde_json::from_str(include_str!("../tauri.conf.json")).expect("tauri.conf.json");
    conf["app"]["security"].clone()
}

/// ディレクティブの値(文字列でも配列でもよい)を、空白区切りのトークンにする。
fn tokens(value: &Value) -> Vec<String> {
    match value {
        Value::String(s) => s.split_whitespace().map(str::to_owned).collect(),
        Value::Array(a) => a.iter().flat_map(tokens).collect(),
        other => panic!("CSP の値が文字列でも配列でもありません: {other}"),
    }
}

fn directives() -> Vec<(String, Vec<String>)> {
    let security = security();
    let csp = security["csp"].as_object().expect("csp が設定されていません");
    csp.iter().map(|(k, v)| (k.clone(), tokens(v))).collect()
}

fn directive(name: &str) -> Vec<String> {
    directives()
        .into_iter()
        .find(|(k, _)| k == name)
        .unwrap_or_else(|| panic!("csp.{name} がありません"))
        .1
}

#[test]
fn script_and_default_sources_are_self_only() {
    assert_eq!(directive("script-src"), ["'self'"]);
    assert_eq!(directive("default-src"), ["'self'"]);
}

/// `script-src-elem` / `script-src-attr` は `script-src` より優先されるので、置かせない。
#[test]
fn script_src_is_not_overridden() {
    for (name, _) in directives() {
        assert!(
            name != "script-src-elem" && name != "script-src-attr",
            "{name} は script-src を上書きします"
        );
    }
}

#[test]
fn eval_is_never_allowed() {
    for (name, values) in directives() {
        for value in values {
            assert!(
                value != "'unsafe-eval'" && value != "'wasm-unsafe-eval'",
                "{name} に {value} があります"
            );
        }
    }
}

#[test]
fn unsafe_inline_is_only_for_styles() {
    for (name, values) in directives() {
        if values.iter().any(|v| v == "'unsafe-inline'") {
            assert_eq!(name, "style-src");
        }
    }
}

#[test]
fn unused_directives_are_closed() {
    for name in ["object-src", "base-uri", "form-action", "frame-ancestors"] {
        assert_eq!(directive(name), ["'none'"], "{name}");
    }
}

/// `http://ipc.localhost` のような特定の接続先はよいが、スキーム全体は許可しない。
#[test]
fn connect_src_has_no_wildcard() {
    for value in directive("connect-src") {
        assert!(
            !["*", "http:", "https:", "ws:", "wss:"].contains(&value.as_str()),
            "connect-src に {value} があります"
        );
    }
}

#[test]
fn only_style_src_modification_is_disabled() {
    assert_eq!(
        security()["dangerousDisableAssetCspModification"],
        serde_json::json!(["style-src"])
    );
}

/// Tauri は `tauri.<platform>.conf.json` を本体の設定に重ねるので、どれも `app.security` を持たせない。
#[test]
fn platform_configs_do_not_override_security() {
    let mut checked = 0;
    for entry in std::fs::read_dir(env!("CARGO_MANIFEST_DIR")).expect("src-tauri") {
        let path = entry.expect("entry").path();
        let name = path.file_name().and_then(|n| n.to_str()).unwrap_or_default();
        if !(name.starts_with("tauri.") && name.ends_with(".conf.json")) || name == "tauri.conf.json" {
            continue;
        }
        let text = std::fs::read_to_string(&path).expect("read config");
        let conf: Value = serde_json::from_str(&text).expect("parse config");
        assert!(conf["app"]["security"].is_null(), "{name} が app.security を上書きします");
        checked += 1;
    }
    assert!(checked >= 1, "tauri.android.conf.json が見つかりません");
}
```

```ts
// HLS / DASH のプレイリストは、動画・音声として扱わない(Issue #612)。Vidstack は、これらを渡されると
// hls.js / dash.js を CDN から読み込む(CSP で遮断される)ので、通常のファイルにする。Vidstack は MIME に
// 関係なく URL の拡張子でも判定するので、URL も見る。
const STREAMING_PLAYLIST_TYPES = new Set(["audio/mpegurl", "audio/x-mpegurl", "video/mpegurl", "video/x-mpegurl"]);
const STREAMING_PLAYLIST_URL = /\.(m3u8|mpd)($|\?)/i;
const isStreamingPlaylist = (f: DriveFile) =>
  STREAMING_PLAYLIST_TYPES.has(f.mimeType) || STREAMING_PLAYLIST_URL.test(f.url);

export const isImage = (f: DriveFile) => f.mimeType.startsWith("image/");
export const isVideo = (f: DriveFile) => f.mimeType.startsWith("video/") && !isStreamingPlaylist(f);
export const isAudio = (f: DriveFile) => f.mimeType.startsWith("audio/") && !isStreamingPlaylist(f);
```

```ts
// CSP 違反の記録(Issue #612)。console.warn に出し、__cspViolations に積む。E2E は各テストの後に
// この配列が空であることを確かめる。WebView が読み込み直されると、それまでの記録は消える。

/// 遮断された画像が並ぶ画面などで増え続けないようにする上限。
export const CSP_VIOLATIONS_LIMIT = 50;

export interface CspViolation {
  directive: string;
  blockedUri: string;
  source: string;
  line: number;
}

type CspTarget = EventTarget & { __cspViolations?: CspViolation[] };

declare global {
  interface Window {
    __cspViolations?: CspViolation[];
  }
}

export function installCspViolationRecorder(target: CspTarget = window): void {
  const list = (target.__cspViolations ??= []);
  target.addEventListener("securitypolicyviolation", (event) => {
    if (list.length >= CSP_VIOLATIONS_LIMIT) return;
    const e = event as SecurityPolicyViolationEvent;
    const v = { directive: e.effectiveDirective, blockedUri: e.blockedURI, source: e.sourceFile, line: e.lineNumber };
    list.push(v);
    console.warn(`[csp] blocked ${v.directive} ${v.blockedUri} (${v.source}:${v.line})`);
  });
}
```

```ts
// 各テストの後に、アプリが CSP 違反を起こしていないことを確かめる(Issue #612)。
// wdio の afterTest は例外を投げてもテストが失敗しないので、mocha のルートフックにする。
// 記録は WebView の読み込み直し(設定のインポートなど)で消えるため、それより前の違反は拾えない。
const SESSION_GONE = /invalid session id|no such window|session (is )?(deleted|terminated|not (created|found))/i;

export const mochaHooks = {
  async afterEach(): Promise<void> {
    let violations: unknown[] | null = null;
    try {
      // 読み込み直しの直後は main.ts がまだ動いていないことがあるので、少し待つ。
      await browser.waitUntil(
        async () => {
          violations = await browser.execute(
            () => (window as unknown as { __cspViolations?: unknown[] }).__cspViolations ?? null,
          );
          return violations !== null;
        },
        { timeout: 5000, timeoutMsg: "CSP violation recorder is not installed" },
      );
    } catch (err) {
      // テストがアプリやセッションを終わらせた場合は確かめようがないので、警告だけにする。
      if (SESSION_GONE.test(String(err))) {
        console.warn(`[csp] skipped the violation check: ${String(err)}`);
        return;
      }
      throw err;
    }
    const found = violations as unknown[] | null;
    if (found && found.length > 0) throw new Error(`CSP violations: ${JSON.stringify(found)}`);
  },
};
```

`e2e/wdio.conf.ts` の `mochaOpts` を次にする(mobile / restart / scheduled の設定は `...base` で引き継ぐ):
